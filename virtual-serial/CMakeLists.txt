cmake_minimum_required(VERSION 3.16)
project(portus_virtual_serial_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(portus_serial_core STATIC core/src/serial_pair.cpp)
target_include_directories(portus_serial_core PUBLIC core/include)
if(MSVC)
  target_compile_options(portus_serial_core PRIVATE /W4 /WX)
else()
  target_compile_options(portus_serial_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
find_package(Threads REQUIRED)
add_executable(serial_pair_tests tests/serial_pair_tests.cpp)
target_link_libraries(serial_pair_tests PRIVATE portus_serial_core Threads::Threads)
add_executable(serial_config_tests tests/serial_config_tests.cpp)
target_link_libraries(serial_config_tests PRIVATE portus_serial_core)
foreach(target serial_pair_tests serial_config_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endif()
endforeach()
# This is the same FIFO implementation compiled into the standalone KMDF driver.
add_executable(labserial_ring_tests standalone-driver/tests/ring_tests.c)
if(MSVC)
  target_compile_options(labserial_ring_tests PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(labserial_ring_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()

option(LABSERIAL_ENABLE_SANITIZERS "ASan and UBSan for Linux user-mode tests" OFF)
if(LABSERIAL_ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR NOT UNIX)
    message(FATAL_ERROR "Sanitizer tests require a Unix GCC/Clang toolchain")
  endif()
  foreach(tgt portus_serial_core serial_pair_tests serial_config_tests labserial_ring_tests)
    target_compile_options(${tgt} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${tgt} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

enable_testing()
add_test(NAME serial_pair_tests COMMAND serial_pair_tests)
add_test(NAME serial_config_tests COMMAND serial_config_tests)
add_test(NAME labserial_ring_tests COMMAND labserial_ring_tests)
